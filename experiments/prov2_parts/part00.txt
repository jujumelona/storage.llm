def relation_name(a,b):
    x=(a,b)
    table={
      ('READ','MUTATE'):'READ_TO_MUTATE',('AUTH','MUTATE'):'AUTH_TO_MUTATE',
      ('MUTATE','VERIFY'):'MUTATE_TO_VERIFY',('VERIFY','FINAL'):'VERIFY_TO_FINAL',
      ('READ','FINAL'):'READ_TO_FINAL',('MUTATE','FINAL'):'MUTATE_TO_FINAL',
      ('ERROR','READ'):'ERROR_TO_RECOVERY',('ERROR','VERIFY'):'ERROR_TO_RECOVERY',
    }
    return table.get(x,f'{a}_TO_{b}')


def relation_features(tr,i,pv):
    e=tr['events'][i];root=pv.roots[i];atoms=pv.pool.atoms(root);rels=[];dist=[];same=0
    for a in atoms:
        j=pv.pool.atom_event.get(a,-1)
        if not (0<=j<i):continue
        s=tr['events'][j];sr=v1.reskey(s);trg=v1.reskey(e)
        sm=int(sr==trg and sr!='none');same+=sm;d=i-j;dist.append(d)
        rels.append((relation_name(s.op,e.op),s.op,s.status,sm,v1.small_bucket(d)))
    two=[]
    for a in atoms:
        j=pv.pool.atom_event.get(a,-1)
        if not (0<=j<i):continue
        for aa in pv.pool.atoms(pv.roots[j]):
            k=pv.pool.atom_event.get(aa,-1)
            if 0<=k<j:
                two.append((tr['events'][k].op,tr['events'][j].op,e.op,int(v1.reskey(tr['events'][k])==v1.reskey(e))))
    cuts=pv.pool.cutsets(root) if pv.pool.value(root) else ()
    mincut=min(cuts,key=lambda x:(len(x),tuple(x))) if cuts else frozenset()
    cutrels=[]
    for a in mincut:
        j=pv.pool.atom_event.get(a,-1)
        if 0<=j<i:cutrels.append(relation_name(tr['events'][j].op,e.op))
    return {'rels':tuple(sorted(rels)),'two':tuple(sorted(set(two))[:6]),'cutrels':tuple(sorted(cutrels)),
            'same':same,'mind':min(dist) if dist else 0,'maxd':max(dist) if dist else 0,
            'srcops':tuple(sorted(set(x[1] for x in rels))),'nrels':len(rels)}


def prov_keys_v2(tr,i,pv):
    old=v1.prov_keys_original(tr,i,pv);rf=relation_features(tr,i,pv);e=tr['events'][i]
    out=list(old)
    for rel,src,st,sm,d in rf['rels']:
        out.append(('EDGE',rel,st,sm,d,e.status))
        out.append(('EDGEOP',src,e.op,sm,d))
    for x in rf['two']:out.append(('PATH2',)+x)
    out += [('EDGESET',rf['srcops'],e.op,v1.small_bucket(rf['nrels']),v1.small_bucket(rf['same'])),
            ('CUTREL',rf['cutrels'],e.op,v1.small_bucket(len(rf['cutrels']))),
            ('EDGEDIST',e.op,v1.small_bucket(rf['mind']),v1.small_bucket(rf['maxd']))]
    return out


def prov_score_v2(tr,i,pm,w,pv):
    f=pv.features[i];rf=relation_features(tr,i,pv)
    vals=[core.loglift(k,pm['total'],pm['positive'],pm['base']) for k in prov_keys_v2(tr,i,pv)]
    emp=sum(sorted(vals,reverse=True)[:6])
    coherent=sum(1 for r,*_ in rf['rels'] if r in {'READ_TO_MUTATE','AUTH_TO_MUTATE','MUTATE_TO_VERIFY','VERIFY_TO_FINAL','READ_TO_FINAL','MUTATE_TO_FINAL'})
    return w['emp']*emp+w['gap']*(1-f['truth'])+w['boundary']*f['boundary']+w['tail']*math.log1p(f['false_tail'])+w['coh']*coherent+w['unique']*int(f['cut']==1 and f['fanout']>=2)

def train_prov(traces):
    n=Counter();p=Counter();N=P=0
    for tr in traces:
        pv=v1.get_prov(tr)
        for i,e in enumerate(tr['events']):
            N+=1;P+=int(e.first_error)
            for k in prov_keys_v2(tr,i,pv):n[k]+=1;p[k]+=int(e.first_error)
    return {'total':n,'positive':p,'base':(P+.7)/(N+1.4)}


def rank01(xs):
    if len(xs)<=1:return [0.]
    order=sorted(range(len(xs)),key=lambda i:(xs[i],-i));r=[0.]*len(xs)
    for j,i in enumerate(order):r[i]=j/(len(xs)-1)
    return r
