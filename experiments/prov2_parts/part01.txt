    anchor=core.predict(tr,m,'product')
    if 0<=anchor<len(br):br[anchor]+=1.0
    return [a+w['mix']*b for a,b in zip(br,pr)]


def predict(tr,m,pm,w,kind):
    if kind in BASE_KINDS:return core.predict(tr,m,kind)
    z=scores(tr,m,pm,w,kind);return max(range(len(z)),key=lambda i:(z[i],-i)) if z else -1

def evaluate(ts,m,pm,w,kind):
    xs=[t for t in ts if t['has_error'] and t['events']];g=[next(i for i,e in enumerate(t['events']) if e.first_error) for t in xs];p=[predict(t,m,pm,w,kind) for t in xs]
    return {'n':len(g),'exact':v1.acc(p,g),'near1':v1.near(p,g),'pred':p,'gold':g,'tids':[t['tid'] for t in xs],'datasets':[t['dataset'] for t in xs]}

def choose_weights(train,val,cfg):
    m=core.train_model(train,cfg);pm=train_prov(train);best=None
    for w in WEIGHTS:
        e=evaluate(val,m,pm,w,'provenance_product');r=evaluate(val,m,pm,w,'provenance_rewired')
        key=(e['exact'],e['near1'],e['exact']-r['exact'],-w['mix'])
        if best is None or key>best[0]:best=(key,w)
    return dict(best[1])


def fold_eval(ts):
    agg={k:{'pred':[],'gold':[],'tids':[],'datasets':[]} for k in KINDS};folds=[]
    for f in range(5):
        test=[t for t in ts if core.hashfold(t['tid'])==f];rest=[t for t in ts if core.hashfold(t['tid'])!=f]
        val=[t for t in rest if core.stable_int('v:'+t['tid'])%5==0];train=[t for t in rest if t not in val]
        cfg=core.choose(train,val);w=choose_weights(train,val,cfg);m=core.train_model(rest,cfg);pm=train_prov(rest);row={'fold':f,'cfg':cfg,'weights':w}
        for k in KINDS:
            e=evaluate(test,m,pm,w,k);row[k]={'n':e['n'],'exact':e['exact'],'near1':e['near1']}
            for q in ('pred','gold','tids','datasets'):agg[k][q]+=e[q]
        folds.append(row)
    met={k:{'n':len(x['gold']),'exact':v1.acc(x['pred'],x['gold']),'near1':v1.near(x['pred'],x['gold'])} for k,x in agg.items()}
    return met,agg,folds


def holds(ts):
    out=[]
    for ds in core.DATASETS:
        test=[t for t in ts if t['dataset']==ds];rest=[t for t in ts if t['dataset']!=ds]
        val=[t for t in rest if core.stable_int('h:'+t['tid'])%5==0];train=[t for t in rest if t not in val]
        cfg=core.choose(train,val);w=choose_weights(train,val,cfg);m=core.train_model(rest,cfg);pm=train_prov(rest);row={'heldout':ds,'cfg':cfg,'weights':w}
        for k in KINDS:
            e=evaluate(test,m,pm,w,k);row[k]={'n':e['n'],'exact':e['exact'],'near1':e['near1']}
        out.append(row)
    return out
