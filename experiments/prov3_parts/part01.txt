        key=(ev['exact'],ev['near1'],ev['exact']-er['exact'],-cfg['margin'],-cfg['support'])
        if best is None or key>best[0]:best=(key,cfg)
    return dict(best[1])


def fold_eval(ts):
    agg={k:{'pred':[],'gold':[],'tids':[],'datasets':[]} for k in KINDS};folds=[]
    for f in range(5):
        test=[t for t in ts if core.hashfold(t['tid'])==f];rest=[t for t in ts if core.hashfold(t['tid'])!=f]
        val=[t for t in rest if core.stable_int('v:'+t['tid'])%5==0];train=[t for t in rest if t not in val]
        bcfg=core.choose(train,val);cfg=choose_cfg(train,val,bcfg);bm=core.train_model(rest,bcfg);lm=train_lifted(rest,cfg)
        row={'fold':f,'base_cfg':bcfg,'lifted_cfg':cfg,'lifted_states':len(lm['states']),'unavoidable':len(lm['unavoidable'])}
        for k in KINDS:
            use=lm if k=='provenance_lifted' or k in BASE_KINDS else train_lifted(rest,cfg,rewire=k.endswith('rewired'),no_resource=k.endswith('no_resource'))
            e=evaluate(test,bm,use,k);row[k]={'n':e['n'],'exact':e['exact'],'near1':e['near1']}
            for q in ('pred','gold','tids','datasets'):agg[k][q]+=e[q]
        folds.append(row)
    met={k:{'n':len(x['gold']),'exact':v1.acc(x['pred'],x['gold']),'near1':v1.near(x['pred'],x['gold'])} for k,x in agg.items()}
    return met,agg,folds


def holdouts(ts):
    rows=[]
    for ds in core.DATASETS:
        test=[t for t in ts if t['dataset']==ds];rest=[t for t in ts if t['dataset']!=ds]
        val=[t for t in rest if core.stable_int('h:'+t['tid'])%5==0];train=[t for t in rest if t not in val]
        bcfg=core.choose(train,val);cfg=choose_cfg(train,val,bcfg);bm=core.train_model(rest,bcfg);lm=train_lifted(rest,cfg)
        row={'heldout':ds,'base_cfg':bcfg,'lifted_cfg':cfg}
        for k in KINDS:
            use=lm if k=='provenance_lifted' or k in BASE_KINDS else train_lifted(rest,cfg,rewire=k.endswith('rewired'),no_resource=k.endswith('no_resource'))
            e=evaluate(test,bm,use,k);row[k]={'n':e['n'],'exact':e['exact'],'near1':e['near1']}
        rows.append(row)
    return rows
