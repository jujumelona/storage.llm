            # Remove one source event from each set for a sequence baseline.
            ce = pv.pool.atom_event[next(iter(cut))]
            re = pv.pool.atom_event[next(iter(random_cut))]
            cut_events = [e for j, e in enumerate(tr["events"]) if j != ce]
            rnd_events = [e for j, e in enumerate(tr["events"]) if j != re]
            cut_bigram.append(bigram_anomaly(cut_events, bg))
            random_bigram.append(bigram_anomaly(rnd_events, bg))
            records.append(
                {
                    "tid": tr["tid"],
                    "dataset": tr["dataset"],
                    "fold": fold,
                    "target": target,
                    "cut_atoms": [pv.pool.atom_name[x] for x in sorted(cut)],
                    "random_atoms": [pv.pool.atom_name[x] for x in sorted(random_cut)],
                    "cut_breaks_target": cut_false,
                    "random_breaks_target": random_false,
                    "cut_downstream_false": ca,
                    "random_downstream_false": ra,
                    "cut_bigram": cut_bigram[-1],
                    "random_bigram": random_bigram[-1],
                }
            )

    def pair_auc(a, b):
        return sum(1.0 if x > y else 0.5 if x == y else 0.0 for x, y in zip(a, b)) / len(a) if a else 0.5

    return {
        "n_pairs": total,
        "minimal_cut_proof_rate": proof_ok / total if total else 0.0,
        "random_same_size_break_rate": random_break / total if total else 0.0,
        "cut_vs_random_downstream_auc": pair_auc(cut_anomaly, random_anomaly),
        "bigram_cut_vs_random_auc": pair_auc(cut_bigram, random_bigram),
        "mean_downstream_margin": statistics.mean(a - b for a, b in zip(cut_anomaly, random_anomaly)) if total else 0.0,
    }, records


def storage_runtime(traces):
    compact = expanded = edge_bytes = roots_count = gates = 0
    sample = []
    for tr in traces:
        pv = get_prov(tr)
        compact += pv.pool.compact_bytes()
        expanded += len(json.dumps([pv.pool.expanded(r) for r in pv.roots], separators=(",", ":")).encode())
        edge_bytes += len(json.dumps(pv.real_dependencies, separators=(",", ":")).encode())
        roots_count += len(pv.roots)
        gates += len(pv.pool.nodes)
        if len(sample) < 3000:
            sample.extend((pv.pool, r) for r in pv.roots[: max(0, 3000 - len(sample))])
    t0 = time.perf_counter()
    memo_value = sum(pool.value(root) for pool, root in sample)
    memo_seconds = time.perf_counter() - t0
    t0 = time.perf_counter()
    raw_value = sum(pool.value_nomemo(root) for pool, root in sample)
    raw_seconds = time.perf_counter() - t0
    assert memo_value == raw_value
    return {
        "traces": len(traces),
        "roots": roots_count,
        "gates": gates,
        "compact_circuit_bytes": compact,
        "expanded_formula_bytes": expanded,
        "raw_dependency_edge_bytes": edge_bytes,
        "compact_vs_expanded_ratio": compact / max(1, expanded),
        "sample_roots": len(sample),
        "memoized_value": memo_value,
        "raw_value": raw_value,
        "outputs_identical": True,
        "memoized_seconds": memo_seconds,
        "recursive_seconds": raw_seconds,
        "evaluation_speedup": raw_seconds / max(memo_seconds, 1e-12),
    }


def serial_counter(counter):
    return [[repr(k), v] for k, v in counter.items()]
